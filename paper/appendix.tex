\section{The Evaluation Semantics of $B$}\label{apx:eval-semantics}

\begin{align*}
  \EE{x}{\gamma} &= \gamma(x) \\
  \EE{\tlet{x}{t_1}{t_2}}{\gamma} &= \EE{t_2}{\gamma, x \mapsto \EE{t_1}{\gamma}} \\
  \EE{\ttrue}{\gamma} &= \strue \\
  \EE{\tfalse}{\gamma} &= \sfalse \\
  \EE{\tif{t_1}{t_2}{t_3}}{\gamma} &=
    \begin{cases}
      \EE{t_2}{\gamma} & \text{if \( \EE{t_{1}}{\gamma} = \strue \)} \\
      \EE{t_3}{\gamma} & \text{otherwise}
    \end{cases} \\
  \EE{\tnil}{\gamma} &= \varepsilon \\
  \EE{\tcons{t_1}{t_2}}{\gamma} &= \scons{\EE{t_1}{\gamma}}{\EE{t_2}{\gamma}} \\
  \EE{\tfoldr{x_1}{x_2}{t_1}{t_2}{t_3}}{\gamma} &= \EEF{x_1}{x_2}{t_1}{t_2}{\gamma}{\EE{t_3}{\gamma}} \\
  \EE{\tlazy{t}}{\gamma} &= \EE{t}{\gamma} \\
  \EE{\tforce{t}}{\gamma} &= \EE{t}{\gamma} \\
  \EE{\ttick{t}}{\gamma} &= \EE{t}{\gamma} \\ \\
  \EEF{x_1}{x_2}{t_1}{t_2}{\gamma}{\snil} &= \EE{t_2}{\gamma} \\
  \EEF{x_1}{x_2}{t_1}{t_2}{\gamma}{\scons{v_1}{v_2}} &= \EE{t_1}{\gamma, x_1 \mapsto v_1, x_2 \mapsto \EEF{x_1}{x_2}{t_1}{t_2}{\gamma}{v_2}}
\end{align*}

\section{The Clairvoyance Semantics of \(\LB\)}\label{apx:clairvoyance-semantics}

\begin{mathpar}
  \inferrule{\gamma(x) = v}{\EC{\gamma}{x}{0}{v}}{\textsc{C-Var}} \qquad
  \inferrule{\EC{\gamma}{t_1}{c_1}{v_1} \\ \EC{\gamma, x \mapsto v_1}{t_2}{c_2}{v_2}}{\EC{\gamma}{\tlet{x}{t_1}{t_2}}{c_1 + c_2}{v_2}}{\textsc{C-Let}} \\
  \inferrule{\\}{\EC{\gamma}{\ttrue}{0}{\strue}}{\textsc{C-True}} \qquad
  \inferrule{\\}{\EC{\gamma}{\tfalse}{0}{\sfalse}}{\textsc{C-False}} \\
  \inferrule{\EC{\gamma}{t_1}{c_1}{\strue} \\ \EC{\gamma}{t_2}{c_2}{v}}{\EC{\gamma}{\tif{t_1}{t_2}{t_3}}{c_1 + c_2}{v}}{\textsc{C-If-True}} \qquad
  \inferrule{\EC{\gamma}{t_1}{c_1}{\sfalse} \\ \EC{\gamma}{t_3}{c_3}{v}}{\EC{\gamma}{\tif{t_1}{t_2}{t_3}}{c_1 + c_3}{v}}{\textsc{C-If-False}} \\
  \inferrule{\\}{\EC{\gamma}{\tnil}{0}{\snil}}{\textsc{C-Nil}} \qquad
  \inferrule{\EC{\gamma}{t_1}{c_1}{v_1} \\ \EC{\gamma}{t_2}{c_1}{v_2}}{\EC{\gamma}{\tcons{t_1}{t_2}}{c_1 + c_2}{\scons{v_1}{v_2}}}{\textsc{C-Cons}} \\
  \inferrule{\EC{\gamma}{t_3}{c_1}{v_1} \\ \ECF{\gamma}{x_1}{x_2}{t_1}{t_2}{v_1}{c_2}{v_2}}{\EC{\gamma}{\tfoldr{x_1}{x_2}{t_1}{t_2}{t_3}}{c_1 + c_2}{v_2}}{\textsc{C-Foldr}} \\
  \inferrule{\EC{\gamma}{t}{c}{v}}{\EC{\gamma}{\tlazy{t}}{c}{\sthunk{v}}}{\textsc{C-Lazy-Result}} \qquad
  \inferrule{\\}{\EC{\gamma}{\tlazy{t}}{0}{\sundefined}}{\textsc{C-Lazy-Undefined}} \\
  \inferrule{\EC{\gamma}{t}{c}{\sthunk{v}}}{\EC{\gamma}{\tforce{t}}{c}{v}}{\textsc{C-Force}} \qquad
  \inferrule{\EC{\gamma}{t}{c}{v}}{\EC{\gamma}{\ttick{t}}{c + 1}{v}}{\textsc{C-Tick}} \\
\end{mathpar}

\section{The Evaluation Semantics of $L$}\label{apx:semantics-logic}

\begin{center}
  \begin{prooftree}
    \hypo{\gamma(x) = v}
    \infer1{\EL{\gamma}{x}{v}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{v_1}}
    \hypo{\EL{\gamma, x \mapsto v_1}{t_2}{v_2}}
    \infer2{\EL{\gamma}{\tlet{x}{t_1}{t_2}}{v_2}}
  \end{prooftree}

  \begin{prooftree}
    \infer0{\EL{\gamma}{\ttrue}{\strue}}
  \end{prooftree}

  \begin{prooftree}
    \infer0{\EL{\gamma}{\tfalse}{\sfalse}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{v}}
    \hypo{\EL{\gamma}{t_2}{v}}
    \infer2{\EL{\gamma}{\teq{t_1}{t_2}}{\strue}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{v_1}}
    \hypo{\EL{\gamma}{t_2}{v_2}}
    \hypo{v_1 \ne v_2}
    \infer3{\EL{\gamma}{\teq{t_1}{t_2}}{\sfalse}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{v_1}}
    \hypo{\EL{\gamma}{t_2}{v_2}}
    \hypo{\sle{v_1}{v_2}}
    \infer3{\EL{\gamma}{\tle{t_1}{t_2}}{\strue}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{v_1}}
    \hypo{\EL{\gamma}{t_2}{v_2}}
    \hypo{\snle{v_1}{v_2}}
    \infer3{\EL{\gamma}{\tle{t_1}{t_2}}{\sfalse}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{\strue}}
    \hypo{\EL{\gamma}{t_2}{v}}
    \infer2{\EL{\gamma}{\tif{t_1}{t_2}{t_3}}{v}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{\sfalse}}
    \hypo{\EL{\gamma}{t_3}{v}}
    \infer2{\EL{\gamma}{\tif{t_1}{t_2}{t_3}}{v}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{v_1}}
    \hypo{\EL{\gamma}{t_2}{v_2}}
    \infer2{\EL{\gamma}{\tpair{t_1}{t_2}}{(v_1, v_2)}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t}{v}}
    \infer1{\EL{\gamma}{\tthunk{t}}{\sthunk{v}}}
  \end{prooftree}

  \begin{prooftree}
    \infer0{\EL{\gamma}{\tundefined}{\sundefined}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{\sthunk{v_1}}}
    \hypo{\EL{\gamma, x \mapsto v_1}{t_2}{v_2}}
    \infer2{\EL{\gamma}{\tcase{t_1}{x}{t_2}{t_3}}{v_2}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{\sundefined}}
    \hypo{\EL{\gamma}{t_3}{v}}
    \infer2{\EL{\gamma}{\tcase{t_1}{x}{t_2}{t_3}}{v}}
  \end{prooftree}

  \begin{prooftree}
    \infer0{\EL{\gamma}{n}{n}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{n_1}}
    \hypo{\EL{\gamma}{t_2}{n_2}}
    \infer2{\EL{\gamma}{\tadd{t_1}{t_2}}{n_1 + n_2}}
  \end{prooftree}

  \begin{prooftree}
    \infer0{\EL{\gamma}{\tnil}{\snil}}
  \end{prooftree}

  \begin{prooftree}
    \hypo{\EL{\gamma}{t_1}{v_1}}
    \hypo{\EL{\gamma}{t_2}{v_2}}
    \infer2{\EL{\gamma}{\tcons{t_1}{t_2}}{\scons{v_1}{v_2}}}
  \end{prooftree}

  \begin{prooftree}
